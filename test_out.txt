"weightedunionfind" : { 
 	 "prefix" : "weightedunionfind",
	"body" : [
		"#include <iostream>",
		"#include <vector>",
		"",
		"using namespace std;",
		"",
		"/**",
		"* @brief 重み付きunion find",
		"* ",
		"* @tparam T 重みの型",
		"*/",
		"template<class T>",
		"struct Weighted_UnionFind {",
		"\tvector<int> par;",
		"\tvector<int> rank;",
		"\tvector<T> diff_weight;",
		"\t/**",
		"\t* @brief Construct a new Weighted_UnionFind object",
		"\t* ",
		"\t* @param n 要素数",
		"\t* @param e 重みの単位元",
		"\t*/",
		"\tWeighted_UnionFind(int n = 1, T e = 0) {",
		"\t\tpar.resize(n);",
		"\t\trank.resize(n);",
		"\t\tdiff_weight.resize(n);",
		"\t\tfor (int i = 0; i < n; i++) {",
		"\t\t\tpar[i] = i;",
		"\t\t\trank[i] = 0;",
		"\t\t\tdiff_weight[i] = e;",
		"\t\t}",
		"\t}",
		"\t/**",
		"\t* @brief 要素 x の根を求める",
		"\t*/",
		"\tint leader(int x) {",
		"\t\tif (par[x] == x) {",
		"\t\t\treturn x;",
		"\t\t} else {",
		"\t\t\tint r = leader(par[x]);",
		"\t\t\tdiff_weight[x] += diff_weight[par[x]];",
		"\t\t\treturn par[x] = r; ",
		"\t\t}",
		"\t}",
		"\t/**",
		"\t* @brief 要素 x の重みを求める",
		"\t*/",
		"\tT weight(int x) {",
		"\t\tleader(x);",
		"\t\treturn diff_weight[x];",
		"\t}",
		"\t/**",
		"\t* @brief 要素 a と要素 b が同じ集合に存在するか",
		"\t*/",
		"\tbool same(int a, int b) {",
		"\t\treturn leader(a) == leader(b);",
		"\t}",
		"\t/**",
		"\t* @brief 要素 x と要素 y を重み w で併合する (weight(y) = weight(x) + w)",
		"\t*/",
		"\tbool merge(int x, int y, T w) {",
		"\t\tw += weight(x);",
		"\t\tw -= weight(y);",
		"\t\tx = leader(x);",
		"\t\ty = leader(y);",
		"\t\tif (x == y) return false;",
		"\t\tif (rank[x] < rank[y]) {",
		"\t\t\tswap(x, y);",
		"\t\t\tw = -w;",
		"\t\t}",
		"\t\tif (rank[x] == rank[y]) {",
		"\t\t\trank[x]++;",
		"\t\t}",
		"\t\tpar[y] = x;",
		"\t\tdiff_weight[y] = w;",
		"\t\treturn true;",
		"\t}",
		"\t/**",
		"\t* @brief 要素 x と要素 y の重みの差 (xとyが同じ集合に属することが前提)",
		"\t* @return T weight(y) - weight(x)",
		"\t*/",
		"\tT diff(int x, int y) {",
		"\t\tassert(leader(x) == leader(y));",
		"\t\treturn weight(y) - weight(x);",
		"\t}",
		"};",
	]
}
